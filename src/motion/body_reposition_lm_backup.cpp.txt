math::Vec3 jointPos(const std::vector<math::Mat4>& models, int joint) {
    return math::Vec3(models[static_cast<size_t>(joint)][3]);
}

// Rotation angle of a unit quaternion away from identity (radians).
float angleOf(math::Quat q) { return 2.0f * std::acos(std::min(1.0f, std::fabs(q.w))); }

// Rotation vector (axis x angle) of a unit quaternion.
math::Vec3 rotationVector(math::Quat q) {
    if (q.w < 0.0f) q = -q;
    const math::Vec3 v(q.x, q.y, q.z);
    const float s = math::length(v);
    return s < 1e-7f ? 2.0f * v : v * (2.0f * std::atan2(s, q.w) / s);
}

// Solves (A + lambda * diag(A)) x = b for a symmetric positive matrix A
// (n x n, row-major) by Cholesky. Returns false if it is not positive.
bool solveDamped(float* a, const float* b, float* x, int n, float lambda) {
    constexpr int kMax = 18;
    float l[kMax * kMax] = {};
    for (int i = 0; i < n; ++i) a[i * n + i] += lambda * std::max(a[i * n + i], 1e-6f);
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j <= i; ++j) {
            float sum = a[i * n + j];
            for (int k = 0; k < j; ++k) sum -= l[i * kMax + k] * l[j * kMax + k];
            if (i == j) {
                if (sum <= 0.0f) return false;
                l[i * kMax + i] = std::sqrt(sum);
            } else {
                l[i * kMax + j] = sum / l[j * kMax + j];
            }
        }
    }
    float y[kMax];
    for (int i = 0; i < n; ++i) {
        float sum = b[i];
        for (int k = 0; k < i; ++k) sum -= l[i * kMax + k] * y[k];
        y[i] = sum / l[i * kMax + i];
    }
    for (int i = n - 1; i >= 0; --i) {
        float sum = y[i];
        for (int k = i + 1; k < n; ++k) sum -= l[k * kMax + i] * x[k];
        x[i] = sum / l[i * kMax + i];
    }
    return true;
}

} // namespace

bool BodyReposition::init(const rig::Rig& rig, const ik::JointLimits& limits, const BodyRepositionJoints& joints,
                        const std::vector<float>& segmentMass, const std::vector<int>& segmentEnd,
                        const std::vector<math::Vec3>& segmentTip) {
    limits_ = &limits;
    segmentMass_ = segmentMass;
    segmentEnd_ = segmentEnd;
    segmentTip_ = segmentTip;
    isLeg_.assign(static_cast<size_t>(rig.numJoints()), 0);
    joints_ = joints;
    if (joints.pelvis < 0 || joints.toes[0] < 0 || joints.toes[1] < 0) return false;
    std::vector<math::Mat4> models(static_cast<size_t>(rig.numJoints()));
    if (!rig.localToModel(rig.restPose(), models)) return false;
    const math::Vec3 pelvis = jointPos(models, joints.pelvis);
    legMass_ = 0.0f;
    for (int i = 0; i < 2; ++i) {
        const ik::TwoBoneChain& c = joints.legs[i];
        const int toe = joints.toes[i];
        // Rest rotations are identity, so rest model offsets are local offsets.
        hipOffset_[i] = jointPos(models, c.start) - pelvis;
        kneeOffset_[i] = jointPos(models, c.mid) - jointPos(models, c.start);
        ankleOffset_[i] = jointPos(models, c.end) - jointPos(models, c.mid);
        toeOffset_[i] = jointPos(models, toe) - jointPos(models, c.end);
        toeTip_[i] = segmentTip[static_cast<size_t>(toe)];
        const int ids[4] = {c.start, c.mid, c.end, toe};
        for (int k = 0; k < 4; ++k) {
            isLeg_[static_cast<size_t>(ids[k])] = 1;
            mass_[i][k] = segmentMass[static_cast<size_t>(ids[k])];
            legMass_ += mass_[i][k];
        }
    }
    ankleHeight_ = jointPos(models, joints.legs[0].end).y;
    toeHeight_ = jointPos(models, joints.toes[0]).y;
    warm_ = false;
    return true;
}

void BodyReposition::legFk(const math::Vec3& pelvisPos, math::Quat pelvisRot, int leg, const State& st, Leg& out) const {
    const math::Quat thigh = pelvisRot * st.q[leg][0];
    const math::Quat shin = thigh * st.q[leg][1];
    const math::Quat foot = shin * st.q[leg][2];
    out.toeRot = foot * st.q[leg][3];
    const math::Vec3 hip = pelvisPos + pelvisRot * hipOffset_[leg];
    out.knee = hip + thigh * kneeOffset_[leg];
    out.ankle = out.knee + shin * ankleOffset_[leg];
    out.toe = out.ankle + foot * toeOffset_[leg];
    out.heel = out.ankle + foot * math::Vec3(0.0f, -ankleHeight_, 0.0f);
    out.ball = out.toe + out.toeRot * math::Vec3(0.0f, -toeHeight_, 0.0f);
    out.tip = out.ball + out.toeRot * toeTip_[leg];
    out.massSum = mass_[leg][0] * 0.5f * (hip + out.knee) + mass_[leg][1] * 0.5f * (out.knee + out.ankle) +
                  mass_[leg][2] * 0.5f * (out.ankle + out.toe) + mass_[leg][3] * (out.toe + out.toeRot * (0.5f * toeTip_[leg]));
}

int BodyReposition::residuals(State& st, math::Vec3& pelvisPos, float* r) const {
    const BodyRepositionSettings& s = *settings_;
    const int p = planted_, o = 1 - planted_;

    // Contact: the planted sole keeps its facing (flat, heel lifted by
    // heelLift); the ankle takes what the shin leaves, within its range. The
    // toes lie flat (as far as the toe joint allows) and the body hangs from
    // the ball of the planted foot.
    const math::Quat footWant = math::fromAxisAngle(math::kRight, st.heelLift);
    const math::Quat shin = st.pelvis * st.q[p][0] * st.q[p][1];
    st.q[p][2] = ik::clampRotation(math::normalize(math::inverse(shin) * footWant), limits_->limit(joints_.legs[p].end));
    const math::Quat foot = shin * st.q[p][2];
    st.q[p][3] = ik::clampRotation(math::normalize(math::inverse(foot)), limits_->limit(joints_.toes[p]));
    Leg legs[2];
    legFk(math::Vec3(0.0f), st.pelvis, p, st, legs[p]);
    pelvisPos = ballGoal_ - legs[p].ball;
    legFk(pelvisPos, st.pelvis, p, st, legs[p]);
    legFk(pelvisPos, st.pelvis, o, st, legs[o]);

    int n = 0;
    auto add = [&](float weight, math::Vec3 v, int count) {
        const float w = std::sqrt(weight);
        for (int i = 0; i < count; ++i) r[n++] = w * v[i];
    };
    add(s.contactWeight, rotationVector(legs[p].toeRot), 3);
    add(s.contactWeight, rotationVector(math::inverse(footWant) * foot), 3);
    add(s.reachWeight, legs[o].ankle - reachGoal_, 3);
    // Balance: centre of mass over the support. The planted foot (heel to toe
    // tip) always; with the reaching foot on the ground, also the ground
    // between the feet (it carries weight as it comes down: contact).
    const math::Vec3 com = (upperMass_ * (pelvisPos + st.pelvis * upperCom_) + legs[0].massSum + legs[1].massSum) /
                           (upperMass_ + legMass_);
    auto flat = [](math::Vec3 v) { return math::Vec2(v.x, v.z); };
    const math::Vec2 c2 = flat(com);
    const math::Vec2 alone = supportExcess(c2, flat(legs[p].heel), flat(legs[p].tip), s.footHalfWidth);
    const math::Vec2 between = supportExcess(c2, 0.5f * (flat(legs[p].heel) + flat(legs[p].tip)),
                                             0.5f * (flat(legs[o].heel) + flat(legs[o].tip)), s.footHalfWidth);
    const math::Vec2 both = math::dot(between, between) < math::dot(alone, alone) ? between : alone;
    const float contact =
        std::min(math::clamp(1.0f - legs[o].heel.y / std::max(s.supportHeight, 1e-3f), 0.0f, 1.0f), reachLoad_);
    const math::Vec2 out = alone + (both - alone) * contact;
    add(s.balanceWeight * balance_, math::Vec3(out.x, out.y, 0.0f), 2);
    // Upright: the pelvis stays level; the legs move first.
    add(s.uprightWeight, rotationVector(st.pelvis), 3);
    // IK hint: each knee points forward (its sideways share in the pelvis frame).
    for (int i = 0; i < 2; ++i) add(s.kneeHintWeight, math::Vec3((st.q[i][0] * math::kForward).x, 0.0f, 0.0f), 1);
    // Comfort: free joints near neutral (the planted ankle is set by the
    // contact; its heel lift counts instead).
    for (int i = 0; i < 2; ++i)
        for (int k = 0; k < 4; ++k)
            add(s.comfortWeight, i == p && k == 2 ? math::Vec3(st.heelLift, 0.0f, 0.0f) : rotationVector(st.q[i][k]), 3);
    return n;
}

void BodyReposition::solve(rig::Pose& pose, const std::vector<math::Mat4>& models, int planted, math::Vec3 plantedAnkle,
                         math::Vec3 reachAnkle, const BodyRepositionSettings& s, float reachLoad, float balance) {
    settings_ = &s;
    balance_ = math::clamp(balance, 0.0f, 1.0f);
    reachLoad_ = math::clamp(reachLoad, 0.0f, 1.0f);
    planted_ = planted;
    reachGoal_ = reachAnkle;
    // The planted ball's spot: where it is when that ankle stands flat at its target.
    ballGoal_ = plantedAnkle + toeOffset_[planted] + math::Vec3(0.0f, -toeHeight_, 0.0f);

    // Upper body (everything above the legs) moves with the pelvis: its mass
    // and centre in the pelvis frame, from the pose as it is.
    const int pelvis = joints_.pelvis;
    math::Vec3 sum(0.0f);
    float total = 0.0f;
    for (size_t j = 0; j < models.size(); ++j) {
        if (isLeg_[j]) continue;
        const float m = segmentMass_[j];
        const int end = segmentEnd_[j];
        const math::Vec3 at = jointPos(models, static_cast<int>(j));
        sum += m * (end >= 0 ? 0.5f * (at + jointPos(models, end))
                             : math::Vec3(models[j] * math::Vec4(0.5f * segmentTip_[j], 1.0f)));
        total += m;
    }
    upperMass_ = total;
    upperCom_ = total > 0.0f ? math::Vec3(math::inverse(models[static_cast<size_t>(pelvis)]) * math::Vec4(sum / total, 1.0f))
                             : math::Vec3(0.0f);

    if (!warm_) {
        state_ = State{};
        for (auto& leg : state_.q)
            for (auto& q : leg) q = math::identityQuat();
        warm_ = true;
    }

    // Damped least squares (Levenberg-Marquardt): every joint axis moves
    // together toward lower error on all constraints at once; each step is
    // clamped to the joint limits (hard).
    struct Dof {
        int leg, k, axis;  // leg -1: pelvis
    };
    Dof dofs[kMaxDofs];
    int n = 0;
    for (int a = 0; a < 3; ++a) dofs[n++] = {-1, 0, a};
    for (int i = 0; i < 2; ++i) {
        for (int a = 0; a < 3; ++a) dofs[n++] = {i, 0, a};  // thigh
        dofs[n++] = {i, 1, 0};                              // knee (hinge about x)
        if (i == planted) {
            dofs[n++] = {i, 4, 0};  // heel lift (the planted ankle and toes follow the contact)
        } else {
            for (int a = 0; a < 3; ++a) dofs[n++] = {i, 2, a};  // ankle
            dofs[n++] = {i, 3, 0};                              // toe
        }
    }
    auto jointOf = [&](int leg, int k) {
        const ik::TwoBoneChain& c = joints_.legs[leg];
        return k == 0 ? c.start : k == 1 ? c.mid : k == 2 ? c.end : joints_.toes[leg];
    };
    const math::Vec3 axes[3] = {math::kRight, math::kUp, math::kForward};
    auto turn = [&](State& st, int d, float angle) {
        const math::Quat q = math::fromAxisAngle(axes[dofs[d].axis], angle);
        if (dofs[d].k == 4) {
            st.heelLift = std::max(st.heelLift + angle, 0.0f);  // the heel cannot go below the ground
        } else if (dofs[d].k == 1) {
            // Knee: stepped as the hip-to-ankle distance (shorter by `angle`,
            // in m), not the knee angle. Near straight, a small bend barely
            // changes that distance, so an angle step could not start to bend
            // a straight leg; a distance step always can.
            const int leg = dofs[d].leg;
            math::Quat& j = st.q[leg][1];
            const float a = math::length(kneeOffset_[leg]), b = math::length(ankleOffset_[leg]);
            const math::Quat q = j.w < 0.0f ? -j : j;
            const float flex = std::max(2.0f * std::atan2(q.x, q.w), 0.0f);
            const float now = std::sqrt(std::max(a * a + b * b + 2.0f * a * b * std::cos(flex), 0.0f));
            const float reach = math::clamp(now - angle, std::fabs(a - b), a + b);
            const float c = math::clamp((reach * reach - a * a - b * b) / (2.0f * a * b), -1.0f, 1.0f);
            j = ik::clampRotation(math::fromAxisAngle(math::kRight, std::acos(c)), limits_->limit(jointOf(leg, 1)));
        } else if (dofs[d].leg < 0) {
            st.pelvis = clampRotationVector(math::normalize(q * st.pelvis), s.pelvisMin, s.pelvisMax);
        } else {
            math::Quat& j = st.q[dofs[d].leg][dofs[d].k];
            j = ik::clampRotation(math::normalize(j * q), limits_->limit(jointOf(dofs[d].leg, dofs[d].k)));
        }
    };
    auto costOf = [](const float* r, int m) {
        float c = 0.0f;
        for (int i = 0; i < m; ++i) c += r[i] * r[i];
        return c;
    };

    math::Vec3 pelvisPos(0.0f);
    float r0[kMaxResiduals], r1[kMaxResiduals], jac[kMaxResiduals * kMaxDofs];
    const int m = residuals(state_, pelvisPos, r0);
    float cost = costOf(r0, m);
    float lambda = 1e-2f;
    constexpr float kEps = 1e-3f;  // rad, finite-difference step
    for (int it = 0; it < s.iterations; ++it) {
        // Jacobian by finite differences (a joint at its limit gives a zero column).
        for (int d = 0; d < n; ++d) {
            State trial = state_;
            turn(trial, d, kEps);
            math::Vec3 at;
            residuals(trial, at, r1);
            for (int i = 0; i < m; ++i) jac[i * kMaxDofs + d] = (r1[i] - r0[i]) / kEps;
        }
        float jtj[kMaxDofs * kMaxDofs], jtr[kMaxDofs], step[kMaxDofs];
        for (int a = 0; a < n; ++a) {
            jtr[a] = 0.0f;
            for (int i = 0; i < m; ++i) jtr[a] -= jac[i * kMaxDofs + a] * r0[i];
            for (int b2 = 0; b2 < n; ++b2) {
                float dot = 0.0f;
                for (int i = 0; i < m; ++i) dot += jac[i * kMaxDofs + a] * jac[i * kMaxDofs + b2];
                jtj[a * n + b2] = dot;
            }
        }
        bool accepted = false;
        for (int tries = 0; tries < 6 && !accepted; ++tries) {
            float sys[kMaxDofs * kMaxDofs];
            std::copy(jtj, jtj + n * n, sys);
            if (!solveDamped(sys, jtr, step, n, lambda)) {
                lambda *= 10.0f;
                continue;
            }
            State trial = state_;
            for (int d = 0; d < n; ++d) turn(trial, d, step[d]);
            math::Vec3 at;
            residuals(trial, at, r1);
            const float c = costOf(r1, m);
            if (c < cost) {
                state_ = trial;
                cost = c;
                pelvisPos = at;
                std::copy(r1, r1 + m, r0);
                lambda = std::max(lambda * 0.3f, 1e-6f);
                accepted = true;
            } else {
                lambda *= 10.0f;
            }
        }
        if (!accepted) break;  // converged
    }
    residuals(state_, pelvisPos, r0);  // refreshes the planted toe

    pose.locals[static_cast<size_t>(pelvis)].position = pelvisPos;
    pose.locals[static_cast<size_t>(pelvis)].rotation = state_.pelvis;
    for (int i = 0; i < 2; ++i)
        for (int k = 0; k < 4; ++k) pose.locals[static_cast<size_t>(jointOf(i, k))].rotation = state_.q[i][k];
}

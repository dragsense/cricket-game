#pragma once
// Body repositioning from the foot targets (hand targets later): the body
// follows wherever the feet are put. Solve from real body constraints only (BIOMECHANICS 3.1):
// finds the pelvis and both legs' joint rotations that best meet, together,
//  - contact: the planted foot's ball rests on its spot (exact: the body
//    hangs from it) and the sole keeps its facing on the ground: it cannot
//    swivel or roll, only the heel may lift about the ball;
//  - joint ranges: every rotation is kept inside its joint limit throughout;
//  (No ground contact yet: collisions and gravity come later.)
//  - balance: the centre of mass over the support: the planted foot, and
//    the ground between both feet as the reaching foot comes down onto it;
//  - reach: the reaching ankle as near its target as the above allow;
//  - comfort: joints near neutral (straight legs, flat heel when nothing
//    asks otherwise).
// The heel lifts by itself: the foot turns about the ball once the ankle runs
// out of range (the lift is free but costs comfort; the ankle follows the shin). Damped least squares (Levenberg-Marquardt) over the
// joint rotations, warm-started from the last solution. Allocates only in init().

#include "ik/joint_limits.h"
#include "ik/limb_ik.h"
#include "math/math.h"
#include "rig/rig.h"

#include <vector>

namespace motion {

struct BodyRepositionSettings {
    float contactWeight = 100.0f;   // toes flat on the planted foot (per rad^2)
    float balanceWeight = 1000.0f;  // centre of mass outside the planted foot (per m^2)
    float reachWeight = 100.0f;     // reaching ankle from its target (per m^2)
    float comfortWeight = 0.1f;     // joint rotation away from neutral (per rad^2)
    float uprightWeight = 10.0f;    // pelvis away from level (per rad^2): the body leans only once the legs run out of range
    float kneeHintWeight = 1.0f;    // IK hint: a knee pointing sideways rather than forward (per unit^2)
    float footHalfWidth = 0.045f;   // m, half the foot's width (support area)
    float supportHeight = 0.1f;     // m; the reaching foot this high above the ground carries no weight
    int iterations = 10;            // solver steps per tick (warm-started)
    math::Vec3 pelvisMin{0.0f}, pelvisMax{0.0f};  // pelvis rotation limits (rotation vector, rad)
};

struct BodyRepositionJoints {
    int pelvis = -1;
    ik::TwoBoneChain legs[2];  // left, right: thigh, shin, foot
    int toes[2] = {-1, -1};
};

class BodyReposition {
public:
    // Reads rest offsets, ground heights and segment masses. Load-time only.
    bool init(const rig::Rig& rig, const ik::JointLimits& limits, const BodyRepositionJoints& joints,
              const std::vector<float>& segmentMass, const std::vector<int>& segmentEnd,
              const std::vector<math::Vec3>& segmentTip);

    // Next solve starts from the rest pose.
    void reset() { warm_ = false; }

    // Solves pelvis and legs in `pose` (character space). `planted` (0 left, 1
    // right) stands with its ankle's rest-flat spot at `plantedAnkle`; the
    // other leg reaches its ankle toward `reachAnkle`. `reachLoad` [0, 1] is
    // the share of weight the reaching foot may still take on the ground (0:
    // its weight has moved off, as before a step lifts it). `balance` [0, 1]
    // scales the centre-of-mass rule: 1 standing still, 0 while stepping (no
    // gravity yet: the body leans only as the feet's places need, so a foot
    // straight under its hip leaves the body upright). `models` holds `pose`
    // in model space (for the upper body's mass). No allocations.
    void solve(rig::Pose& pose, const std::vector<math::Mat4>& models, int planted, math::Vec3 plantedAnkle,
               math::Vec3 reachAnkle, const BodyRepositionSettings& s, float reachLoad = 1.0f, float balance = 1.0f);

private:
    struct State {
        math::Quat pelvis = math::identityQuat();
        math::Quat q[2][4];  // per leg: thigh, shin, foot, toe (local rotations)
        float heelLift = 0.0f;  // planted foot: turn about the ball, heel up (rad)
    };
    struct Leg {
        math::Vec3 knee, ankle, toe, heel, ball, tip;  // points; heel, ball, tip on the sole
        math::Quat toeRot;
        math::Vec3 massSum;  // sum of segment mass x centre
    };

    static constexpr int kMaxDofs = 18;
    static constexpr int kMaxResiduals = 48;

    void legFk(const math::Vec3& pelvisPos, math::Quat pelvisRot, int leg, const State& st, Leg& out) const;
    // Weighted errors of all constraints into `r`; derives the planted toe and
    // the pelvis position. Returns the count.
    int residuals(State& st, math::Vec3& pelvisPos, float* r) const;

    const ik::JointLimits* limits_ = nullptr;
    BodyRepositionJoints joints_;
    std::vector<float> segmentMass_;
    std::vector<int> segmentEnd_;
    std::vector<math::Vec3> segmentTip_;
    std::vector<char> isLeg_;  // per joint: a leg segment (solved here)
    math::Vec3 hipOffset_[2]{}, kneeOffset_[2]{}, ankleOffset_[2]{}, toeOffset_[2]{}, toeTip_[2]{};
    float ankleHeight_ = 0.0f, toeHeight_ = 0.0f;  // rest heights above the ground
    float mass_[2][4]{};                             // thigh, shin, foot, toe
    float legMass_ = 0.0f;

    // Per solve inputs.
    int planted_ = 1;
    math::Vec3 ballGoal_{0.0f}, reachGoal_{0.0f};
    float reachLoad_ = 1.0f;
    float balance_ = 1.0f;
    math::Vec3 upperCom_{0.0f};  // upper body's centre of mass in the pelvis frame
    float upperMass_ = 0.0f;
    const BodyRepositionSettings* settings_ = nullptr;

    State state_;
    bool warm_ = false;
};

} // namespace motion
